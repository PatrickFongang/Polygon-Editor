PolygonBuilder::PolygonBuilder() {
    currentPolygon = std::make_unique<Polygon>();
}

PolygonBuilder& PolygonBuilder::addVertex(float x, float y, Continuity continuity) {
    auto newVertex = std::make_unique<Vertex>(sf::Vector2f(x, y));
    newVertex->setContinuity(continuity);
    
    Vertex* vertexPtr = newVertex.get(); 
    currentPolygon->addVertex(std::move(newVertex)); 

    if (!firstVertex) {
        firstVertex = vertexPtr;
    } else {
        auto newEdge = std::make_unique<LineEdge>(lastVertex, vertexPtr);
        currentPolygon->addEdge(std::move(newEdge));
    }
    
    lastVertex = vertexPtr;
    return *this;
}

PolygonBuilder& PolygonBuilder::withLastEdgeAsBezier() {
    if (!currentPolygon->getEdges().empty()) {
        auto& edges = currentPolygon->getEdges();
        Vertex* v1 = edges.back()->getVertex1();
        Vertex* v2 = edges.back()->getVertex2();

        // change the last edge to BezierEdge
        currentPolygon->replaceLastEdge(std::make_unique<BezierEdge>(v1, v2));
    }
    return *this;
}

PolygonBuilder& PolygonBuilder::withLastEdgeAsArc() {
    if (!currentPolygon->getEdges().empty()) {
        auto& edges = currentPolygon->getEdges();
        Vertex* v1 = edges.back()->getVertex1();
        Vertex* v2 = edges.back()->getVertex2();

        // change the last edge to ArcEdge
        currentPolygon->replaceLastEdge(std::make_unique<ArcEdge>(v1, v2));
    }
    return *this;
}


PolygonBuilder& PolygonBuilder::closePolygon() {
    if (firstVertex && lastVertex && firstVertex != lastVertex) {
        auto closingEdge = std::make_unique<LineEdge>(lastVertex, firstVertex);
        
        currentPolygon->addEdge(std::move(closingEdge));
    }
    return *this;
}

std::unique_ptr<Polygon> PolygonBuilder::build() {
    return std::move(currentPolygon);
}

void BezierEdge::draw(const Drawer& drawer) const {
    const int N = 50; 
    const float dt = 1.0f / N;

    sf::Vector2f P0(v1->getPosition().x, v1->getPosition().y);
    sf::Vector2f P1(controlPoint1->getPosition().x, controlPoint1->getPosition().y);
    sf::Vector2f P2(controlPoint2->getPosition().x, controlPoint2->getPosition().y);
    sf::Vector2f P3(v2->getPosition().x, v2->getPosition().y);

    // calculate the coefficients for the cubic Bezier curve
    sf::Vector2f a = -P0 + 3.0f * P1 - 3.0f * P2 + P3;
    sf::Vector2f b =  3.0f * P0 - 6.0f * P1 + 3.0f * P2;
    sf::Vector2f c = -3.0f * P0 + 3.0f * P1;
    sf::Vector2f d =  P0;

    // incremental drawing using forward differences
    float dt2 = dt * dt;
    float dt3 = dt2 * dt;

    sf::Vector2f currentPos = d;

    sf::Vector2f delta1 = a * dt3 + b * dt2 + c * dt;
    sf::Vector2f delta2 = 6.0f * a * dt3 + 2.0f * b * dt2;
    sf::Vector2f delta3 = 6.0f * a * dt3;

    sf::Vector2f prevPos = currentPos;

    for (int i = 1; i <= N; ++i) {
        currentPos += delta1;
        delta1 += delta2;
        delta2 += delta3;

        drawer.drawLine(prevPos.x, prevPos.y, currentPos.x, currentPos.y);
        
        prevPos = currentPos;
    }
}

#include <cmath>

void ArcEdge::draw(const Drawer& drawer) const {
    sf::Vector2f p1(v1->getPosition().x, v1->getPosition().y);
    sf::Vector2f p2(v2->getPosition().x, v2->getPosition().y);

    sf::Vector2f center = (p1 + p2) / 2.0f;

    sf::Vector2f radiusVec = p1 - center;
    
    float radius = std::sqrt(radiusVec.x * radiusVec.x + radiusVec.y * radiusVec.y);

    float startAngle = std::atan2(radiusVec.y, radiusVec.x);
    float endAngle = startAngle + M_PI;

    const int segments = 30; 
    float angleStep = (endAngle - startAngle) / segments;

    sf::Vector2f prevPos = p1;

    for (int i = 1; i <= segments; ++i) {
        float angle = startAngle + i * angleStep;
        
        sf::Vector2f currentPos(
            center.x + radius * std::cos(angle),
            center.y + radius * std::sin(angle)
        );

        drawer.drawLine(prevPos.x, prevPos.y, currentPos.x, currentPos.y);

        prevPos = currentPos;
    }
}

#include "edge/line_edge.hpp"

void Polygon::splitEdge(size_t edgeIndex, const sf::Vector2f &newVertexPosition, const sf::Color &newVertexColor, Continuity newVertexContinuity)
{
    if (edgeIndex >= edges.size())
    {
        return;
    }

    Edge *edgeToSplit = edges[edgeIndex].get();
    Vertex *v2 = edgeToSplit->getVertex2();

    auto splitVertex = std::make_unique<Vertex>(newVertexPosition, newVertexColor, newVertexContinuity);
    edgeToSplit->setVertex2(splitVertex.get());

    auto newEdge = std::make_unique<LineEdge>(splitVertex.get(), v2);
    edges.insert(edges.begin() + edgeIndex + 1, std::move(newEdge));
    vertices.push_back(std::move(splitVertex));
}

void Polygon::removeVertex(size_t vertexIndex)
{
    if (vertexIndex >= vertices.size())
    {
        return;
    }

    Vertex *vertexToRemove = vertices[vertexIndex].get();
    auto [edge1, edge2] = findEdgePair(vertexToRemove);

    if (edge1 && edge2)
    {
        edge1->setVertex2(edge2->getVertex2());
        std::erase_if(edges, [edge2](const std::unique_ptr<Edge> &e)
                      { return e.get() == edge2; });
        vertices.erase(vertices.begin() + vertexIndex);
    }
}

void Polygon::replaceLastEdge(std::unique_ptr<Edge> newEdge) {
    if (!edges.empty()) {
        edges.back() = std::move(newEdge);
    }
}

std::pair<Edge *, Edge *> Polygon::findEdgePair(Vertex *middleVertex)
{
    Edge *edge1 = nullptr;
    Edge *edge2 = nullptr;

    for (const auto &edge : edges)
    {
        if (edge->getVertex2() == middleVertex)
        {
            edge1 = edge.get();
        }
        else if (edge->getVertex1() == middleVertex)
        {
            edge2 = edge.get();
        }
    }

    return {edge1, edge2};
}

    void removeVertex(size_t vertexIndex);
    void replaceLastEdge(std::unique_ptr<Edge> newEdge);

    {
        vertices.push_back(std::make_unique<Vertex>(sf::Vector2f(x, y)));
    }

    {
        edges.push_back(std::move(edge));
    }

    {
        return vertices;
    }

    const std::vector<std::unique_ptr<Edge>> &getEdges() const
    {
        return edges;
    }

    {
        if (index < vertices.size())
        {
            return vertices[index].get();
        }
        return nullptr;
    }

    {
        if (index < edges.size())
        {
            return edges[index].get();
        }
        return nullptr;
    }

    {
        return vertices.size();
    }
    const size_t getEdgeCount() const
    {
        return edges.size();
    }

#include <SFML/Graphics.hpp>
#include <SFML/System/Clock.hpp>
#include <SFML/Window/Event.hpp>

#include "imgui.h"
#include "imgui-SFML.h"

#include "scene/scene.hpp"
#include "ui/ui_manager.hpp"
#include "ui/input_handler.hpp"

int main() {
    // initialize SFML window
    sf::ContextSettings settings;
    settings.antialiasingLevel = 4;
    sf::RenderWindow window(sf::VideoMode(1280, 800), "Edytor Wielokatow - Projekt GK", sf::Style::Default, settings);
    window.setFramerateLimit(60);

    // initialize ImGui-SFML
    if (!ImGui::SFML::Init(window)) {
        return -1;
    }

    // initialize the scene with a default drawer
    auto initialDrawer = std::make_unique<LibraryDrawer>(window);
    Scene scene(std::move(initialDrawer));
    scene.initPredefinedScene();

    UIManager uiManager;
    InputHandler inputHandler;
    
    bool showDocs = false; 

    sf::Clock deltaClock;

    // main loop
    while (window.isOpen()) {
        sf::Event event;
        while (window.pollEvent(event)) {
            ImGui::SFML::ProcessEvent(window, event);

            if (event.type == sf::Event::Closed) {
                window.close();
            }

            bool imguiCapturesMouse = ImGui::GetIO().WantCaptureMouse;
            bool imguiCapturesKeyboard = ImGui::GetIO().WantCaptureKeyboard;

            if (!imguiCapturesMouse && !imguiCapturesKeyboard) {
                inputHandler.handleEvent(event, window, scene, showDocs);
            }
        }

        ImGui::SFML::Update(window, deltaClock.restart());

        uiManager.renderUI(scene, window, showDocs);

        window.clear(sf::Color(30, 30, 35)); 
        
        scene.render(window);                
        ImGui::SFML::Render(window);         

        window.display();
    }

    ImGui::SFML::Shutdown();
    return 0;
}

void BresenhamDrawer::drawPixel(int x, int y) const
{
    sf::Vertex pixel(sf::Vector2f(x, y), sf::Color::White);
    window.draw(&pixel, 1, sf::Points);
}

void BresenhamDrawer::drawLine(float x1, float y1, float x2, float y2) const
{
    int ix1 = static_cast<int>(x1);
    int iy1 = static_cast<int>(y1);
    int ix2 = static_cast<int>(x2);
    int iy2 = static_cast<int>(y2);

    int dx = std::abs(ix2 - ix1);
    int dy = std::abs(iy2 - iy1);
    int sx = (ix1 < ix2) ? 1 : -1;
    int sy = (iy1 < iy2) ? 1 : -1;
    int err = dx - dy;

    while (true)
    {
        drawPixel(ix1, iy1);
        if (ix1 == ix2 && iy1 == iy2)
            break;
        int e2 = 2 * err;
        if (e2 > -dy)
        {
            err -= dy;
            ix1 += sx;
        }
        if (e2 < dx)
        {
            err += dx;
            iy1 += sy;
        }
    }
}
